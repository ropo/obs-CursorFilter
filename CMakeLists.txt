cmake_minimum_required(VERSION 3.20)
project(obs_cursor_filter VERSION 0.1.0 LANGUAGES C CXX)

set(OBS_SOURCE_DIR "" CACHE PATH "OBS Studio ソースのパス")
set(OBS_BUILD_DIR "" CACHE PATH "OBS Studio ビルドのパス")
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h" OR NOT EXISTS "${OBS_BUILD_DIR}/libobs/Release/obs.lib")
  message(FATAL_ERROR "OBS_SOURCE_DIR と OBS_BUILD_DIR にビルド済みの OBS Studio を指定してください")
endif()

add_library(obs-cursor-filter MODULE src/filter.c src/cursor-capture.c src/cur-file.cpp)
target_compile_features(obs-cursor-filter PRIVATE c_std_11)
if(MSVC)
  target_compile_options(obs-cursor-filter PRIVATE /utf-8 /W4)
endif()
target_compile_definitions(obs-cursor-filter PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_include_directories(obs-cursor-filter PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_BUILD_DIR}/config" "${OBS_BUILD_DIR}/libobs")
target_link_libraries(obs-cursor-filter PRIVATE "${OBS_BUILD_DIR}/libobs/Release/obs.lib" user32 gdi32 advapi32 shcore windowscodecs ole32 shlwapi)
set_target_properties(obs-cursor-filter PROPERTIES PREFIX "")
install(TARGETS obs-cursor-filter RUNTIME DESTINATION obs-cursor-filter LIBRARY DESTINATION obs-cursor-filter)

option(CURSOR_FILTER_BUILD_TESTS "回帰テストをビルドする" OFF)
if(CURSOR_FILTER_BUILD_TESTS)
  add_executable(cursor-filter-test tests/window-deadlock.c src/cursor-capture.c src/cur-file.cpp)
  target_compile_features(cursor-filter-test PRIVATE c_std_11)
  target_compile_definitions(cursor-filter-test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_compile_options(cursor-filter-test PRIVATE /utf-8 /W4)
  target_include_directories(cursor-filter-test PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_BUILD_DIR}/config" "${OBS_BUILD_DIR}/libobs")
  target_link_libraries(cursor-filter-test PRIVATE "${OBS_BUILD_DIR}/libobs/Release/obs.lib" user32 gdi32 advapi32 shcore windowscodecs ole32 shlwapi)
  add_executable(cursor-render-test tests/cursor-render.c src/cursor-capture.c src/cur-file.cpp)
  target_compile_features(cursor-render-test PRIVATE c_std_11)
  target_compile_definitions(cursor-render-test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
  target_compile_options(cursor-render-test PRIVATE /utf-8 /W4)
  target_include_directories(cursor-render-test PRIVATE "${OBS_SOURCE_DIR}/libobs" "${OBS_BUILD_DIR}/config" "${OBS_BUILD_DIR}/libobs")
  target_link_libraries(cursor-render-test PRIVATE "${OBS_BUILD_DIR}/libobs/Release/obs.lib" user32 gdi32 advapi32 shcore windowscodecs ole32 shlwapi)
endif()
